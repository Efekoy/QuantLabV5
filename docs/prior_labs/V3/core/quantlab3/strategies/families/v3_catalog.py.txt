"""The V3 strategy catalog (NEW in V3): hypotheses H01-H50 of the V3 brief plus executor
hypotheses X1-X3. Every family emits UP-type and DOWN-type events evaluated at a bar's CLOSE;
`mode` maps them to trades (continuation: up->long; reversal: up->short). Fills are at the next
bar's open (engine). Notation: Z(n) = n-bar move / (same-time-of-day sigma(n) * sqrt(n));
PE(n) = path efficiency; D = mean RTH range of the previous 10 sessions; ATRh = ATR(14) of
completed 15-minute bars. Grids live in config/search_space.yaml; default_grid is a tiny
test grid only.
"""
from __future__ import annotations

import numpy as np

from quantlab3.features import indicators as ind
from quantlab3.strategies.families.v3_common import (B, T, V3Family, armed_trigger, breakout_accept,
                                                      extreme_rejection, failed_break, first_in_session, ge,
                                                      impulse_pullback, in_sm, le, onset, shift_in_session)

RTH_S, RTH_E = T("09:30"), T("16:00")


def _cond(p, key, fn_true, n):
    """All-True if the confluence condition is absent (None)."""
    if p.get(key) is None:
        return np.ones(n, dtype=bool), np.ones(n, dtype=bool)
    return fn_true()


def _rth(fc):
    return in_sm(fc, RTH_S, RTH_E)


def _dirz(fc, x, thr):
    return ge(x, thr), le(x, -thr)


# =============================================================================== A. momentum

class EfficientMomentum(V3Family):
    name = "v3_h01_efficient_momentum"
    hypothesis, section = "H01", "A"
    title = "Efficient momentum"
    summary = "Z(n) crosses +/-z; confluence: normalised path efficiency PE(n)*sqrt(n) >= pe."
    conditions = ("pe",)
    default_grid = {"n": [30], "z": [1.5], "pe": [None, 0.4], "mode": ["continuation"]}

    def events(self, fc, p):
        n = int(p["n"])
        u, d = _dirz(fc, fc.Z(n), float(p["z"]))
        if p["pe"] is not None:
            e = ge(fc.pe(n) * np.sqrt(n), float(p["pe"]))
            u, d = u & e, d & e
        return onset(u), onset(d)

    def event_text(self, p):
        e = "" if p["pe"] is None else (f" and its path efficiency |net|/sum|1-min changes| over those bars, "
                                        f"x sqrt({p['n']}), is >= {p['pe']}")
        return (f"the {p['n']}-minute move first exceeds +{p['z']} sigma_tod(n)*sqrt(n){e}",
                f"the {p['n']}-minute move first falls below -{p['z']} sigma_tod(n)*sqrt(n){e}")


class MomentumAcceleration(V3Family):
    name = "v3_h02_momentum_acceleration"
    hypothesis, section = "H02", "A"
    title = "Momentum acceleration"
    summary = ("Recent k-bar impulse Z(k) >= zk; confluence acc: the preceding m-bar move agrees in "
               "direction (acc=0) and recent per-minute speed >= acc x prior per-minute speed.")
    conditions = ("acc",)
    tied = {"acc": ("m",)}
    default_grid = {"k": [5], "zk": [1.5], "m": [20], "acc": [None, 1.5], "mode": ["continuation"]}

    def events(self, fc, p):
        k, m = int(p["k"]), int(p["m"])
        u, d = _dirz(fc, fc.Z(k), float(p["zk"]))
        if p["acc"] is not None:
            rk = fc.ret(k)
            rp = ind.shift_same_seg(fc.ret(m), k, fc.seg)       # move over [t-k-m, t-k]
            vr = rk / k
            vp = rp / m
            a = float(p["acc"])
            with np.errstate(invalid="ignore"):
                cu = (vp > 0) & (vr >= a * vp) if a > 0 else (vp > 0)
                cd = (vp < 0) & (vr <= a * vp) if a > 0 else (vp < 0)
            fin = np.isfinite(vp) & np.isfinite(vr)
            u, d = u & cu & fin, d & cd & fin
        return onset(u), onset(d)

    def event_text(self, p):
        base = f"the {p['k']}-minute move first exceeds {p['zk']} sigma units"
        if p["acc"] is None:
            x = ""
        elif p["acc"] == 0:
            x = f", and the preceding {p['m']}-minute move was in the same direction"
        else:
            x = (f", the preceding {p['m']}-minute move was in the same direction, and the recent per-minute "
                 f"speed is >= {p['acc']}x the preceding per-minute speed")
        return base + " (up)" + x, base.replace("exceeds", "exceeds (down)") + x


class MultiHorizonMomentum(V3Family):
    name = "v3_h03_multi_horizon"
    hypothesis, section = "H03", "A"
    title = "Multi-horizon momentum"
    summary = "Directional agreement of Z(5), Z(30), Z(120) (each >= z) for the selected horizons."
    conditions = ("h5", "h30", "h120")
    need_one_condition = True
    default_grid = {"h5": [True], "h30": [None, True], "h120": [True], "z": [0.5], "mode": ["continuation"]}

    def events(self, fc, p):
        z = float(p["z"])
        n = fc.v.n
        u = np.ones(n, bool)
        d = np.ones(n, bool)
        for h in (5, 30, 120):
            if p.get(f"h{h}"):
                a, b = _dirz(fc, fc.Z(h), z)
                u &= a
                d &= b
        return onset(u), onset(d)

    def event_text(self, p):
        hs = [str(h) for h in (5, 30, 120) if p.get(f"h{h}")]
        return (f"the {'/'.join(hs)}-minute moves are all >= +{p['z']} sigma units (first bar)",
                f"the {'/'.join(hs)}-minute moves are all <= -{p['z']} sigma units (first bar)")


class HTFTrendImpulse(V3Family):
    name = "v3_h04_htf_trend_impulse"
    hypothesis, section = "H04", "A"
    title = "HTF trend + short-term impulse"
    summary = "HTF state Z(htf) >= zh (2h or 4h) and/or a k-minute impulse Z(k) >= zi, same direction."
    conditions = ("zh", "zi")
    need_one_condition = True
    tied = {"zh": ("htf",), "zi": ("k",)}
    default_grid = {"htf": [120], "zh": [None, 0.5], "k": [5], "zi": [1.5], "mode": ["continuation"]}

    def events(self, fc, p):
        n = fc.v.n
        u = np.ones(n, bool)
        d = np.ones(n, bool)
        if p["zh"] is not None:
            a, b = _dirz(fc, fc.Z(int(p["htf"])), float(p["zh"]))
            u, d = u & a, d & b
        if p["zi"] is not None:
            a, b = _dirz(fc, fc.Z(int(p["k"])), float(p["zi"]))
            u, d = u & a, d & b
        return onset(u), onset(d)

    def event_text(self, p):
        parts = []
        if p["zh"] is not None:
            parts.append(f"the {p['htf']}-minute trend Z >= {p['zh']}")
        if p["zi"] is not None:
            parts.append(f"the {p['k']}-minute impulse Z >= {p['zi']}")
        s = " and ".join(parts)
        return s + " (up; first bar)", s.replace(">=", "<= -") + " (down; first bar)"

class CompressionExpansion3(V3Family):
    name = "v3_h06_compression_expansion"
    hypothesis, section = "H06", "A"
    title = "Compression to expansion"
    summary = ("Close breaks the previous N-bar high/low; confluence: the N bars before were compressed "
               "(rv_N / time-of-day norm <= cth); optional strength Z(5) >= 1.")
    conditions = ("cth", "strong")
    default_grid = {"N": [30], "cth": [None, 0.7], "strong": [None], "mode": ["continuation"]}

    def events(self, fc, p):
        N = int(p["N"])
        c = fc.arr("c")
        u = ge(c - fc.hi_prev(N), 1e-12)
        d = le(c - fc.lo_prev(N), -1e-12)
        if p["cth"] is not None:
            s = le(fc.compression(N), float(p["cth"]))
            u, d = u & s, d & s
        if p["strong"] is not None:
            a, b = _dirz(fc, fc.Z(5), float(p["strong"]))
            u, d = u & a, d & b
        return onset(u), onset(d)

    def event_text(self, p):
        x = "" if p["cth"] is None else f"; the previous {p['N']} bars' realised vol was <= {p['cth']}x its time-of-day norm"
        y = "" if p["strong"] is None else f"; the 5-minute move >= {p['strong']} sigma units"
        return (f"the close breaks above the previous {p['N']}-bar high{x}{y}",
                f"the close breaks below the previous {p['N']}-bar low{x}{y}")


class OpeningDrive(V3Family):
    name = "v3_h07_opening_drive"
    hypothesis, section = "H07", "A"
    title = "Opening-drive continuation"
    summary = "At 09:30+w: RTH move >= m x D; confluence: efficiency since the open >= e."
    conditions = ("e",)
    default_grid = {"w": [15], "m": [0.15], "e": [None, 0.5], "mode": ["continuation"]}

    def events(self, fc, p):
        ro, rh, rl, rp, *_ = fc.rth()
        c = fc.arr("c")
        at = fc.at_time(RTH_S + int(p["w"]))
        mv = (c - ro) / fc.D()
        u = at & ge(mv, float(p["m"]))
        d = at & le(mv, -float(p["m"]))
        if p["e"] is not None:
            with np.errstate(invalid="ignore", divide="ignore"):
                ef = ge(np.abs(c - ro) / rp, float(p["e"]))
            u, d = u & ef, d & ef
        return B(u), B(d)

    def event_text(self, p):
        e = "" if p["e"] is None else f" with path efficiency since the open >= {p['e']}"
        return (f"at 09:30+{p['w']}min the close is >= {p['m']} x D above the RTH open{e}",
                f"at 09:30+{p['w']}min the close is >= {p['m']} x D below the RTH open{e}")

    name = "v3_h08_morning_trend"
    hypothesis, section = "H08", "A"

class LastHourTrend(MorningTrend):
    name = "v3_h09_last_hour_trend"
    hypothesis, section = "H09", "A"
    title = "Last-hour trend continuation"
    summary = "At 15:00: session return from the open >= m x D; confluences: efficiency, closing location."
    default_grid = {"cut": ["15:00"], "m": [0.3], "e": [None], "cl": [None, 0.8], "mode": ["continuation"]}


def day_trend_events(fc, cut_sm, m, e, cl):
    ro, rh, rl, rp, *_ = fc.rth()
    c = fc.arr("c")
    at = fc.at_time(cut_sm)
    mv = (c - ro) / fc.D()
    u = at & ge(mv, m)
    d = at & le(mv, -m)
    if e is not None:
        with np.errstate(invalid="ignore", divide="ignore"):
            ef = ge(np.abs(c - ro) / rp, float(e))
        u, d = u & ef, d & ef
    if cl is not None:
        with np.errstate(invalid="ignore", divide="ignore"):
            loc = (c - rl) / (rh - rl)
        u, d = u & ge(loc, float(cl)), d & le(loc, 1.0 - float(cl))
    return B(u), B(d)


def trend_text(p, side):
    s = "above" if side == "up" else "below"
    x = f"at {p['cut']} the close is >= {p['m']} x D {s} the RTH open"
    if p.get("e") is not None:
        x += f", path efficiency since the open >= {p['e']}"
    if p.get("cl") is not None:
        x += (f", close in the top {round((1 - p['cl']) * 100)}% of the RTH range" if side == "up"
              else f", close in the bottom {round((1 - p['cl']) * 100)}% of the RTH range")
    return x


class ImpulsePullback(V3Family):
    name = "v3_h10_impulse_pullback"
    hypothesis, section = "H10", "A"
    title = "Impulse + pullback continuation"
    summary = ("Impulse Z(L) >= zi, then a controlled pullback of >= retr of the impulse (cancelled beyond "
               "75% or after 60 bars), then resumption: close beyond the previous bar's extreme. "
               "retr None = enter at the impulse (parent).")
    conditions = ("retr",)
    default_grid = {"L": [15], "zi": [1.5], "retr": [None, 0.33], "mode": ["continuation"]}

    qual = 0

    def events(self, fc, p):
        L = int(p["L"])
        u, d = _dirz(fc, fc.Z(L), float(p["zi"]))
        r = -1.0 if p["retr"] is None else float(p["retr"])
        q = {None: 0, "fast": 1, "slow": 2}[p.get("qual")]
        return impulse_pullback(fc.arr("c"), fc.arr("h"), fc.arr("l"), onset(u), onset(d), L,
                                np.asarray(fc.arr("sday")), np.asarray(fc.seg), r, 0.75, 60, q)

    def event_text(self, p):
        base = f"a {p['L']}-minute impulse >= {p['zi']} sigma units"
        if p["retr"] is None:
            return base + " (up) begins", base + " (down) begins"
        q = {None: "", "fast": " reached within 5 bars of the extreme (FAST)",
             "slow": " taking >= 15 bars from the extreme (SLOW)"}[p.get("qual")]
        x = (f" is followed by a pullback of >= {p['retr']} of the impulse (<= 0.75){q} and then a close "
             f"beyond the previous bar's extreme in the impulse direction")
        return base + " (up)" + x, base + " (down)" + x


class FailedCountertrend(V3Family):
    name = "v3_h11_failed_countertrend"
    hypothesis, section = "H11", "A"
    title = "Failed countertrend move"
    summary = ("Trend: Z(120) measured before the counter move >= zt; counter move: 10-minute Z <= -zc; "
               "resumption within 30 bars: close above the previous r-bar high. zc None = resumption "
               "trigger in trend without a counter move (parent).")
    conditions = ("zc",)
    default_grid = {"zt": [0.75], "zc": [None, 1.0], "r": [3], "mode": ["continuation"]}

    def events(self, fc, p):
        r = int(p["r"])
        c = fc.arr("c")
        tu = gt0(c - fc.hi_prev(r))
        td = gt0(fc.lo_prev(r) - c)
        zt = float(p["zt"])
        if p["zc"] is None:
            su, sd = _dirz(fc, fc.Z(120), zt)
            return onset(su & tu), onset(sd & td)
        trend_u, trend_d = _dirz(fc, fc.Zlag(120, 10), zt)
        cu, cd = _dirz(fc, fc.Z(10), float(p["zc"]))
        arm_u = onset(trend_u & cd)     # up-trend, counter move down
        arm_d = onset(trend_d & cu)
        return armed_trigger(arm_u, arm_d, B(tu), B(td), np.asarray(fc.arr("sday")), 30)

    def event_text(self, p):
        if p["zc"] is None:
            return (f"with the 120-minute Z >= {p['zt']}, the close first breaks above the previous {p['r']}-bar high",
                    f"with the 120-minute Z <= -{p['zt']}, the close first breaks below the previous {p['r']}-bar low")
        return (f"the 120-minute Z before a 10-minute counter move <= -{p['zc']} sigma units was >= {p['zt']}; within "
                f"30 bars the close breaks above the previous {p['r']}-bar high",
                f"the 120-minute Z before a 10-minute counter move >= +{p['zc']} sigma units was <= -{p['zt']}; within "
                f"30 bars the close breaks below the previous {p['r']}-bar low")


def gt0(x):
    return ge(x, 1e-12)


class BreakoutAcceptance(V3Family):
    name = "v3_h12_breakout_acceptance"
    hypothesis, section = "H12", "A"
    title = "Breakout + acceptance"
    summary = ("Level: previous 120-bar high/low, overnight high/low, previous-day RTH high/low. "
               "Acceptance: first close beyond (parent), 2nd / 3rd consecutive close, retest, displacement.")
    default_grid = {"level": ["pd"], "acc": ["touch", "retest"], "mode": ["continuation"]}
    ACC = {"touch": 0, "close2": 1, "close3": 2, "retest": 3, "displace": 4}

    def levels(self, fc, lv):
        if lv == "roll120":
            return fc.hi_prev(120), fc.lo_prev(120)
        s = fc.session()
        if lv == "on":
            return s.on_hi, s.on_lo
        return s.p_hi, s.p_lo

    def events(self, fc, p):
        up, dn = self.levels(fc, p["level"])
        rth = _rth(fc)
        up = np.where(rth, up, np.nan)
        dn = np.where(rth, dn, np.nan)
        return breakout_accept(fc.arr("c"), fc.arr("h"), fc.arr("l"), up, dn, fc.atrh(),
                               np.asarray(fc.arr("sday")), self.ACC[p["acc"]], 30)

    def event_text(self, p):
        lv = {"roll120": "previous 120-bar high", "on": "overnight high", "pd": "previous RTH high"}[p["level"]]
        a = {"touch": "the first close above it", "close2": "the second consecutive close above it",
             "close3": "the third consecutive close above it",
             "retest": "after the break, a retest to within 0.25 ATRh that holds, then a close above the previous bar's high (within 30 bars)",
             "displace": "after the break, a close >= 0.5 ATRh above it (within 30 bars)"}[p["acc"]]
        return (f"RTH break of the {lv}: {a}",
                f"RTH break of the {lv.replace('high', 'low')}: {a.replace('above', 'below').replace('high', 'low')}")

class DecelerationExhaustion(V3Family):
    name = "v3_h14_deceleration_exhaustion"
    hypothesis, section = "H14", "B"
    title = "Deceleration exhaustion"
    summary = ("Extended move Z(N) >= z and a new N-bar high, while the last-5-bar per-minute speed "
               "<= r x the per-minute speed of the earlier part of the move. r None = no deceleration (parent).")
    conditions = ("r",)
    default_grid = {"N": [30], "z": [1.5], "r": [None, 0.5], "mode": ["reversal"]}

    def events(self, fc, p):
        N = int(p["N"])
        u, d = _dirz(fc, fc.Z(N), float(p["z"]))
        u &= ge(fc.arr("h") - fc.hi_prev(N), 1e-12)
        d &= le(fc.arr("l") - fc.lo_prev(N), -1e-12)
        if p["r"] is not None:
            vr = fc.ret(5) / 5.0
            ve = ind.shift_same_seg(fc.ret(N - 5), 5, fc.seg) / (N - 5)
            r = float(p["r"])
            with np.errstate(invalid="ignore"):
                du = (ve > 0) & (vr <= r * ve)
                dd = (ve < 0) & (vr >= r * ve)
            u, d = u & du, d & dd
        return onset(u), onset(d)

    def event_text(self, p):
        x = "" if p["r"] is None else (f", while the last 5 minutes' speed is <= {p['r']}x the speed of the "
                                       f"preceding {int(p['N']) - 5} minutes")
        return (f"the {p['N']}-minute move >= {p['z']} sigma units and a new {p['N']}-bar high{x}",
                f"the {p['N']}-minute move <= -{p['z']} sigma units and a new {p['N']}-bar low{x}")

class _FailedLevel(V3Family):
    natural = True
    level_name = ""

    def lv(self, fc, p):
        raise NotImplementedError

    def events(self, fc, p):
        up, dn = self.lv(fc, p)
        ok = in_sm(fc, RTH_S, T("15:30"))
        return failed_break(fc.arr("h"), fc.arr("l"), fc.arr("c"), up, dn, fc.atrh(), np.asarray(fc.arr("sday")),
                            B(ok), int(p["k"]), float(p["depth"]))

    def abs_target(self, fc, p, fill):

        mid = (up + dn) / 2.0

        return (f"the first RTH trade above the {self.level_name} high (reaching >= {p['depth']} ATRh beyond) is "
                f"followed within {p['k']} bars by a close back below it",
                f"the first RTH trade below the {self.level_name} low (reaching >= {p['depth']} ATRh beyond) is "
                f"followed within {p['k']} bars by a close back above it")

class LargeMorningReversal(V3Family):
    name = "v3_h19_large_morning_reversal"
    hypothesis, section = "H19", "B"
    title = "Large morning move reversal"
    summary = ("RTH high - open >= m x D, then (10:00-13:00) a retracement from the high of >= dr of that "
               "move (deterioration). dr None = fade at 11:00 without deterioration (parent). Natural target: RTH open.")
    conditions = ("dr",)
    natural = True
    default_grid = {"m": [0.5], "dr": [None, 0.25], "mode": ["reversal"]}

    def events(self, fc, p):
        ro, rh, rl, *_ = fc.rth()
        c = fc.arr("c")
        D = fc.D()
        m = float(p["m"])
        if p["dr"] is None:
            at = fc.at_time(T("11:00"))
            return B(at & ge((c - ro) / D, m)), B(at & le((c - ro) / D, -m))
        dr = float(p["dr"])
        w = in_sm(fc, T("10:00"), T("13:00"))
        with np.errstate(invalid="ignore", divide="ignore"):
            cu = w & ge((rh - ro) / D, m) & ge((rh - c) / (rh - ro), dr)
            cd = w & ge((ro - rl) / D, m) & ge((c - rl) / (ro - rl), dr)
        return first_in_session(cu, fc), first_in_session(cd, fc)

    def abs_target(self, fc, p, fill):
        ro = fc.rth()[0]
        return ro, ro

    def event_text(self, p):
        if p["dr"] is None:
            return (f"at 11:00 the close is >= {p['m']} x D above the RTH open",
                    f"at 11:00 the close is >= {p['m']} x D below the RTH open")
        return (f"between 10:00 and 13:00, with the RTH high >= {p['m']} x D above the open, price first retraces "
                f">= {p['dr']} of that move from the high",
                f"between 10:00 and 13:00, with the RTH low >= {p['m']} x D below the open, price first retraces "
                f">= {p['dr']} of that move from the low")

class SMT(_Cross):
    name = "v3_h26_smt_nonconfirmation"
    hypothesis = "H26"
    title = "Mechanical SMT / non-confirmation"
    summary = ("One market makes a new extreme (previous N-bar or RTH session high/low) that the other does "
               "not. kind nq_only: this market swept, other did not; es_only: the other swept, this did not. "
               "Confirmation 'rej': within 10 bars this market closes beyond its previous 3-bar extreme "
               "against the sweep. Fade.")
    conditions = ("confirm",)
    default_grid = {"level": ["roll60"], "kind": ["nq_only"], "confirm": [None, "rej"], "mode": ["reversal"]}

    def events(self, fc, p):
        h, l = fc.arr("h"), fc.arr("l")
        om = fc.v.other
        oh = np.where(om.valid, om.h, np.nan)
        ol = np.where(om.valid, om.l, np.nan)
        sday = np.asarray(fc.arr("sday"))
        if p["level"] == "session":
            ro, rh, rl, *_ = fc.rth()
            orh, orl = fc.o_rth()
            xh, xl = shift_in_session(rh, sday), shift_in_session(rl, sday)
            yh, yl = shift_in_session(orh, sday), shift_in_session(orl, sday)
            rth_ok = in_sm(fc, T("10:00"), RTH_E)
        else:
            n = int(p["level"][4:])
            xh, xl = fc.hi_prev(n), fc.lo_prev(n)
            yh, yl = fc.o_hi_prev(n), fc.o_lo_prev(n)
            rth_ok = _rth(fc)
        with np.errstate(invalid="ignore"):
            x_up = np.isfinite(xh) & (h > xh)
            x_dn = np.isfinite(xl) & (l < xl)
            y_up = np.isfinite(yh) & np.isfinite(oh) & (oh > yh)
            y_dn = np.isfinite(yl) & np.isfinite(ol) & (ol < yl)
            y_noup = np.isfinite(yh) & np.isfinite(oh) & (oh <= yh)
            y_nodn = np.isfinite(yl) & np.isfinite(ol) & (ol >= yl)
            x_noup = np.isfinite(xh) & (h <= xh)
            x_nodn = np.isfinite(xl) & (l >= xl)
        if p["kind"] == "nq_only":
            du, dd = x_up & y_noup, x_dn & y_nodn
        else:
            du, dd = y_up & x_noup, y_dn & x_nodn
        du, dd = onset(du & rth_ok), onset(dd & rth_ok)
        if p["confirm"] is None:
            return du, dd
        c = fc.arr("c")
        tu = gt0(fc.lo_prev(3) - c)        # bearish rejection after an upside sweep
        td = gt0(c - fc.hi_prev(3))
        return armed_trigger(du, dd, B(tu), B(td), sday, 10)

    def event_text(self, p):
        who = "this market" if p["kind"] == "nq_only" else "the other index future"
        oth = "the other" if p["kind"] == "nq_only" else "this market"
        lv = "the RTH session high (after 10:00)" if p["level"] == "session" else f"its previous {p['level'][4:]}-bar high"
        c = "" if p["confirm"] is None else " and then, within 10 bars, this market closes below its previous 3-bar low"
        u = f"{who} trades above {lv} while {oth} does not{c}"
        return u, u.replace("above", "below").replace("high", "low").replace("closes below its previous 3-bar low",
                                                                             "closes above its previous 3-bar high")

def _three_windows(fc, w):
    r3 = fc.ret(w)
    r2 = ind.shift_same_seg(r3, w, fc.seg)
    r1 = ind.shift_same_seg(r3, 2 * w, fc.seg)
    return r1, r2, r3

class TimeSinceExtreme(V3Family):
    name = "v3_h36_time_since_extreme"
    hypothesis, section = "H36", "D"
    title = "Time since session high/low"
    summary = ("stale: the RTH high was set >= Tm minutes ago and the close is still in the top 20% of the "
               "RTH range (failed to extend). fresh: a new RTH high after >= Tm minutes without one. "
               "Each tested as continuation and reversal (mirror for lows). After 10:30.")
    default_grid = {"Tm": [60], "kind": ["stale"], "mode": ["continuation", "reversal"]}

    def events(self, fc, p):
        ro, rh, rl, rp, sh, sl, nb, cx = fc.rth()
        c, h, l = fc.arr("c"), fc.arr("h"), fc.arr("l")
        sday = np.asarray(fc.arr("sday"))
        Tm = int(p["Tm"])
        ok = in_sm(fc, T("10:30"), T("15:30"))
        if p["kind"] == "stale":
            with np.errstate(invalid="ignore", divide="ignore"):
                loc = (c - rl) / (rh - rl)
            u = ok & (sh >= Tm) & ge(loc, 0.8)
            d = ok & (sl >= Tm) & le(loc, 0.2)

class EarlyTrendDay(V3Family):
    name = "v3_h45_early_trend_day"
    hypothesis, section = "H45", "F"
    title = "Early trend-day classification"
    summary = ("At the cutoff count five transparent criteria: |return from open| >= 0.2D, RTH range >= 0.45D, "
               "efficiency >= 0.2, close location in the direction >= 0.75, 30-minute realised vol >= its "
               "time-of-day norm. Trend day if >= k are met; trade the open-to-cutoff direction.")
    default_grid = {"cut": ["10:30"], "k": [4], "mode": ["continuation"]}

    def events(self, fc, p):

        c = fc.arr("c")
        D = fc.D()
        at = fc.at_time(T(p["cut"]))
        with np.errstate(invalid="ignore", divide="ignore"):
            r = (c - ro) / D
            loc = (c - rl) / (rh - rl)
            n = (ge(np.abs(r), TD_RET).astype(int) + ge((rh - rl) / D, TD_RANGE).astype(int)
                 + ge(np.abs(c - ro) / rp, TD_EFF).astype(int) + ge(fc.vol_shock(30), TD_VOL).astype(int))
            nu = n + ge(loc, TD_LOC).astype(int)
            nd = n + le(loc, 1 - TD_LOC).astype(int)

        return B(at & (nu >= k) & gt0(r)), B(at & (nd >= k) & gt0(-r))

class EarlyRangeDay(V3Family):
    name = "v3_h46_early_range_day"
    hypothesis, section = "H46", "F"
    title = "Early range-day classification + fade"
    summary = ("At the cutoff count three criteria: efficiency since open <= 0.1, >= 4 crossings of the RTH "
               "open, |return| <= 0.12D. Range day if >= strict are met (None = no classification, parent). "
               "Then until 15:00 fade the first bar that trades beyond the cutoff's RTH high (low) and closes "
               "back inside. Natural target: mid of the cutoff range.")
    conditions = ("strict",)
    natural = True
    default_grid = {"cut": ["11:00"], "strict": [None, 2], "mode": ["reversal"]}

    def _parts(self, fc, p):
        ro, rh, rl, rp, sh, sl, nb, cx = fc.rth()
        cut = T(p["cut"])
        c = fc.arr("c")
        D = fc.D()
        with np.errstate(invalid="ignore", divide="ignore"):
            n = (le(np.abs(c - ro) / rp, RD_EFF).astype(np.float64) + (cx >= RD_CROSS).astype(np.float64)
                 + le(np.abs(c - ro) / D, RD_RET).astype(np.float64))
        ncut = fc.val_at(f"rd_n_{p['cut']}", lambda: n, cut)
        mh = fc.val_at("rth_hi", lambda: rh, cut)
        ml = fc.val_at("rth_lo", lambda: rl, cut)
        return ncut, mh, ml

    def events(self, fc, p):
        ncut, mh, ml = self._parts(fc, p)
        c, h, l = fc.arr("c"), fc.arr("h"), fc.arr("l")
        ok = in_sm(fc, T(p["cut"]), T("15:00"))
        u = ok & ge(h - mh, 1e-12) & gt0(mh - c)
        d = ok & le(l - ml, -1e-12) & gt0(c - ml)
        if p["strict"] is not None:
            cl = ge(ncut, float(p["strict"]))
            u, d = u & cl, d & cl
        return first_in_session(u, fc), first_in_session(d, fc)

    def abs_target(self, fc, p, fill):
        _, mh, ml = self._parts(fc, p)
        m = (mh + ml) / 2
        return m, m

    def event_text(self, p):
        x = "" if p["strict"] is None else f" on a day classified at {p['cut']} as a range day (>= {p['strict']} of 3 criteria)"
        return (f"after {p['cut']} (until 15:00) a bar trades above the {p['cut']} RTH high and closes back below it{x}",
                f"after {p['cut']} (until 15:00) a bar trades below the {p['cut']} RTH low and closes back above it{x}")


def _gap(fc):
    s = fc.session()
    ro = fc.rth()[0]
    with np.errstate(invalid="ignore", divide="ignore"):
        return (ro - s.p_cl) / fc.D()

class CorrRegimeMomentum(_Cross):
    name = "v3_x3_corr_regime_momentum"
    hypothesis, section = "X3", "X"
    title = "Correlation-regime momentum"
    summary = ("Executor hypothesis X3: momentum Z(30) >= z is market-wide (and should persist) when NQ/ES "
               "one-minute correlation over the last 390 bars is high (>= 0.9), idiosyncratic when low (<= 0.75).")
    conditions = ("reg",)
    default_grid = {"z": [1.5], "reg": [None, "high"], "mode": ["continuation"]}

    def events(self, fc, p):
        u, d = _dirz(fc, fc.Z(30), float(p["z"]))
        if p["reg"] is not None:
            cr = fc.corr(390)
            s = ge(cr, 0.9) if p["reg"] == "high" else le(cr, 0.75)
            u, d = u & s, d & s
        return onset(u), onset(d)

    def event_text(self, p):
        x = {None: "", "high": " with 390-minute NQ/ES correlation >= 0.9", "low": " with 390-minute NQ/ES correlation <= 0.75"}[p["reg"]]
        return (f"the 30-minute move first exceeds +{p['z']} sigma units{x}", f"the 30-minute move first falls below -{p['z']} sigma units{x}")
